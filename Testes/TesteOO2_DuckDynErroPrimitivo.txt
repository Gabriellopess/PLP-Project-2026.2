// dyn representa referencias a objetos: atribuir um valor primitivo a uma
// variavel dyn e erro de tipo.
// Resultado: erro-tipo
{
  classe Pato {
    string nome = "";
    Pato(string nome) { this.nome := nome },
    proc falar() { write(this.nome ++ ": Quack!") }
  };
  {
    dyn d = 5;
    write(d)
  }
}

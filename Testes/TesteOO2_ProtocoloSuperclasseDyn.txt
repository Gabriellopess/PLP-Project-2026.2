// Mesmo programa de TesteOO2_ProtocoloSuperclasse, mas a referencia passa
// por uma variavel dyn (o "cast" da linguagem). O compilador aceita, e na
// execucao a classe real (Marreco) satisfaz Anfibio (regra R7).
// Saida esperada:
//   Patinhas nadando
{
  protocolo Anfibio {
    proc nadar()
  },
  classe Ave {
    string nome = "";
    Ave(string nome) { this.nome := nome },
    proc falar() { write(this.nome ++ ": piu") }
  },
  classe Marreco extends Ave {
    int x = 0;
    Marreco(string nome) { this.nome := nome },
    proc nadar() { write(this.nome ++ " nadando") }
  },
  classe Lago {
    int visitas = 0;
    Lago() { skip },
    proc receber(Anfibio a) {
      this.visitas := this.visitas + 1;
      a.nadar()
    }
  };
  {
    Lago l := new Lago(),
    Ave a := new Marreco("Patinhas");
    {
      dyn d = a;
      l.receber(d)
    }
  }
}

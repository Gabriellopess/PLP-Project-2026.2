// Exemplo 9 do README: de dyn para um tipo verificado (regra R7).
// Passar um dyn para o parametro Falante e aceito na compilacao; na
// execucao, o interpretador confere que a classe real (Pedra) nao
// satisfaz Falante.
// Saida esperada:
//   ErroDuckTypingException: argumento 1 de ouvir: Pedra não satisfaz Falante
{
  protocolo Falante {
    proc falar()
  },
  classe Pedra {
    int peso = 0;
    Pedra(int peso) { this.peso := peso },
    proc rolar() { write("rolando") }
  },
  classe Plateia {
    int ouvidos = 0;
    Plateia() { skip },
    proc ouvir(Falante f) {
      this.ouvidos := this.ouvidos + 1;
      f.falar()
    }
  };
  {
    Plateia pl := new Plateia(),
    dyn d := new Pedra(10);
    pl.ouvir(d)
  }
}
